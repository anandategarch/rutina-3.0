   187→
   188→function WeeklyDailyCharts({ workouts }: { workouts: Workout[] }) {
   189→  const today = todayStr();
   190→  const days = React.useMemo(() => {
   191→    const labels = rangeDays(addDays(today, -6), today);
   192→    const agg = labels.map((d) => ({ label: dayName(d, true), km: 0, tonnage: 0, kcal: 0, minutes: 0 }));
   193→    const idx = new Map(labels.map((d, i) => [d, i]));
   194→    for (const w of workouts) {
   195→      const i = idx.get(w.date);
   196→      if (i === undefined) continue;
   197→      agg[i].minutes += w.durationMin;
   198→      agg[i].km += w.distanceKm ?? 0;
   199→      agg[i].kcal += w.calories ?? 0;
   200→      if (w.type === "strength") {
   201→        for (const it of parseStrength(w.strength)) agg[i].tonnage += it.weight * it.reps * it.sets;
   202→      }
   203→    }
   204→    return agg;
   205→  }, [workouts, today]);
   206→
   207→  const charts = [
   208→    { title: "Jarak kardio", unit: "km", digits: 1, color: SAGE, values: days.map((d) => Math.round(d.km * 10) / 10) },
   209→    { title: "Tonase beban", unit: "kg", digits: 0, color: SAGE_DEEP, values: days.map((d) => Math.round(d.tonnage)) },
   210→    { title: "Kalori", unit: "kkal", digits: 0, color: SAGE, values: days.map((d) => Math.round(d.kcal)) },
   211→    { title: "Durasi", unit: "menit", digits: 0, color: SAGE_DEEP, values: days.map((d) => Math.round(d.minutes)) },
   212→  ];
   213→
   214→  return (
   215→    <div className="mt-3 border-t border-border/60 pt-3">
   216→      <p className="mb-2 text-[11px] font-bold uppercase tracking-wide text-muted-foreground">Pola 7 hari terakhir</p>
   217→      <div className="grid grid-cols-2 gap-2">
   218→        {charts.map((c) => (
   219→          <div key={c.title} className="rounded-xl bg-secondary/50 p-2.5">
   220→            <div className="mb-1 flex items-baseline justify-between gap-1">
   221→              <span className="truncate text-[10px] font-semibold text-muted-foreground">{c.title}</span>
   222→              <span className="shrink-0 text-[11px] font-bold tabular">
   223→                {formatNumber(c.values[6] ?? 0, c.digits)}
   224→                <span className="ml-0.5 text-[9px] font-semibold text-muted-foreground"> {c.unit}</span>
   225→              </span>
   226→            </div>
   227→            <MiniBars data={days.map((d, i) => ({ label: d.label, value: c.values[i] }))} height={48} color={c.color} highlightIndex={6} />
   228→          </div>
   229→        ))}
   230→      </div>
   231→    </div>
   232→  );
   233→}

   267→function WorkoutTab({
   268→  workouts,
   269→  loading,

   271→}: {

   273→  loading: boolean;
   274→  checkins: CheckIn[];
   275→}) {
   276→  const openQuickAction = useRutinaStore((s) => s.openQuickAction);
   277→  const today = todayStr();
   278→  const wStart = weekStart(today);
   279→
   280→  const weekWorkouts = workouts.filter((w) => w.date >= wStart && w.date <= today);
   281→
   282→  const summary = React.useMemo(() => {
   283→    let cardioKm = 0;
   284→    let totalMin = 0;
   285→    let calories = 0;
   286→    let tonnage = 0;
   287→    const days = new Set<string>();
   288→    for (const w of weekWorkouts) {
   289→      days.add(w.date);
   290→      totalMin += w.durationMin;
   291→      if (w.distanceKm != null) cardioKm += w.distanceKm;
   292→      if (w.calories != null) calories += w.calories;
   293→      if (w.type === "strength") {
   294→        for (const it of parseStrength(w.strength)) {
   295→          tonnage += it.weight * it.reps * it.sets;
   296→        }
   297→      }
   298→    }
   299→    return { sessions: weekWorkouts.length, cardioKm, totalMin, calories, tonnage, activeDays: days.size };
   300→  }, [weekWorkouts]);
   301→
   302→  // ===== Coach: skor kesiapan =====
   303→  const todayCheckin = checkins.find((c) => c.date === today);
   304→  const refCheckin = todayCheckin ?? checkins[0] ?? null;
   305→
   306→  let readiness: number | null = null;
   307→  let readinessLabel = "";
   308→  if (refCheckin) {
   309→    const sleepHours = refCheckin.sleepHours ?? 7;
   310→    const soreness = refCheckin.soreness ?? 3;
   311→    const stress = refCheckin.stress ?? 3;
   312→    const energy = refCheckin.energy ?? 3;
   313→    readiness = clampScore(sleepHours * 12 + (5 - soreness) * 10 + (5 - stress) * 8 + energy * 10);
   314→    readinessLabel =
   315→      readiness >= 75 ? "Mantap — gas maksimal" : readiness >= 50 ? "Wajar — intensitas moderat" : "Kurang — recovery dulu";
   316→  }
   317→
   318→  // ===== Coach: deteksi deload sederhana =====
   319→  const deloadExercise = React.useMemo(() => {
   320→    const strengthWorkouts = workouts
   321→      .filter((w) => w.type === "strength")
   322→      .sort((a, b) => b.date.localeCompare(a.date));
   323→    const byEx = new Map<string, number[]>(); // max weight per sesi, urut terbaru dulu
   324→    for (const w of strengthWorkouts) {
   325→      const byName = new Map<string, number>();
   326→      for (const it of parseStrength(w.strength)) {
   327→        byName.set(it.exercise, Math.max(byName.get(it.exercise) ?? 0, it.weight));
   328→      }
   329→      byName.forEach((maxW, name) => {
   330→        const arr = byEx.get(name) ?? [];
   331→        arr.push(maxW);
   332→        byEx.set(name, arr);
   333→      });
   334→    }
   335→    for (const [name, occ] of byEx) {
   336→      if (occ.length >= 3 && occ[0] <= occ[2]) return name;
   337→    }
   338→    return null;
   339→  }, [workouts]);
   340→
   341→  const cardioWeekCount = weekWorkouts.filter((w) => w.type !== "strength").length;
   342→  const varietyWarning = cardioWeekCount >= 5;
   343→
   344→  if (loading) return <TabSkeleton />;
   345→
   346→  return (
   347→    <>
   348→      {/* Ringkasan mingguan */}
   349→      <SectionCard title="Ringkasan Pekan Ini" tone="studio" icon={<Timer className="h-4 w-4 text-studio-deep" />}>
   350→        <div className="grid grid-cols-2 gap-3">

   356→          <StatTile label="Kalori" value={formatNumber(summary.calories)} sub="kkal terbakar" icon={<Flame className="h-3.5 w-3.5" />} tone="studio" />
   357→        </div>

   482→function E1rmCard({ workouts }: { workouts: Workout[] }) {
   483→  const top3 = React.useMemo(() => {
   484→    const best = new Map<string, { name: string; muscleGroup: string; e1rm: number }>();
   485→    const history = new Map<string, number[]>();
   486→    const dated = workouts.filter((w) => w.type === "strength").sort((a, b) => a.date.localeCompare(b.date));
   487→    for (const w of dated) {
   488→      const byName = new Map<string, { e1rm: number; muscleGroup: string }>();
   489→      for (const it of parseStrength(w.strength)) {
   490→        const val = it.weight * (1 + it.reps / 30);
   491→        const cur = byName.get(it.exercise);
   492→        if (!cur || val > cur.e1rm) byName.set(it.exercise, { e1rm: val, muscleGroup: it.muscleGroup });
   493→      }
   494→      byName.forEach((v, name) => {
   495→        const arr = history.get(name) ?? [];
   496→        arr.push(v.e1rm);
   497→        history.set(name, arr);
   498→        const b = best.get(name);
   499→        if (!b || v.e1rm > b.e1rm) best.set(name, { name, muscleGroup: v.muscleGroup, e1rm: v.e1rm });
   500→      });
   501→    }
   502→    return [...best.values()]
   503→      .sort((a, b) => b.e1rm - a.e1rm)
   504→      .slice(0, 3)
   505→      .map((r) => ({ ...r, history: history.get(r.name) ?? [] }));
   506→  }, [workouts]);
   507→

   577→function classifyMuscleGroup(raw: string): MuscleGroupKey {
   578→  const s = raw.toLowerCase();
   579→  for (const g of MUSCLE_GROUPS) {

   581→  }

   585→function MuscleBalanceCard({ workouts }: { workouts: Workout[] }) {
